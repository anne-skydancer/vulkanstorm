cmake_minimum_required(VERSION 3.24)
project(SoftwareVulkanTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
get_filename_component(VIEWER_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
list(APPEND CMAKE_MODULE_PATH "${VIEWER_ROOT}/indra/cmake")
set(ADDRESS_SIZE 64)
set(WINDOWS ${WIN32})
if(UNIX AND NOT APPLE)
  set(LINUX ON)
endif()
set(USE_DILIGENTCORE ON)
set(ENV{LL_BUILD} "software-vulkan-tests")
if(NOT AUTOBUILD_INSTALL_DIR)
  message(FATAL_ERROR "Pass the viewer's Autobuild install directory")
endif()
set(TEST_PACKAGES "${AUTOBUILD_INSTALL_DIR}")
include(Linking)
# Variables.cmake chooses the viewer build's packages directory. The standalone
# tool must instead consume exactly the install directory supplied by its caller.
set(AUTOBUILD_INSTALL_DIR "${TEST_PACKAGES}")
include(DiligentCore)
add_executable(diligent_render_test diligent_render_test.cpp)
enable_testing()
add_test(NAME device_selection COMMAND diligent_render_test --test-selection)
target_link_libraries(diligent_render_test PRIVATE ll::diligentcore ll::vulkan)
if(WIN32)
  target_link_libraries(diligent_render_test PRIVATE user32)
  target_compile_definitions(diligent_render_test PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
else()
  find_package(SDL2 REQUIRED)
  find_library(X11_XCB_LIBRARY X11-xcb REQUIRED)
  target_link_libraries(diligent_render_test PRIVATE SDL2::SDL2 ${X11_XCB_LIBRARY} ${CMAKE_DL_LIBS})
  set_target_properties(diligent_render_test PROPERTIES BUILD_RPATH "$ORIGIN")
endif()
foreach(runtime ${DILIGENTCORE_RUNTIME_FILES})
  add_custom_command(TARGET diligent_render_test POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "${DILIGENTCORE_RUNTIME_DIR}/${runtime}" "$<TARGET_FILE_DIR:diligent_render_test>")
endforeach()
