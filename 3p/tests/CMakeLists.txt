cmake_minimum_required(VERSION 3.24)
project(VulkanGhiSmoke LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT GHI_PACKAGES)
  message(FATAL_ERROR "Set GHI_PACKAGES to the Autobuild installation directory")
endif()
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../../indra/cmake")
set(ADDRESS_SIZE 64)
# The dependency test does not use viewer build flags.
set(LL_BUILD_ENV "dependency-smoke")
include(Linking)
set(AUTOBUILD_INSTALL_DIR "${GHI_PACKAGES}")
set(USE_DILIGENTCORE ON)
include(DiligentCore)
add_executable(vulkan_ghi_smoke factory.cpp)
target_link_libraries(vulkan_ghi_smoke PRIVATE ll::diligentcore ll::vulkan)
if(MSVC)
  target_compile_options(vulkan_ghi_smoke PRIVATE /EHsc)
endif()
foreach(runtime ${DILIGENTCORE_RUNTIME_FILES})
  add_custom_command(TARGET vulkan_ghi_smoke POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "${DILIGENTCORE_RUNTIME_DIR}/${runtime}" "$<TARGET_FILE_DIR:vulkan_ghi_smoke>")
endforeach()
set_target_properties(vulkan_ghi_smoke PROPERTIES BUILD_RPATH "$ORIGIN")
enable_testing()
add_test(NAME vulkan_ghi_factory COMMAND vulkan_ghi_smoke)
